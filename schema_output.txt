--- TABLE: questions ---
CREATE TABLE `questions` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `type` enum('enem','concurso') COLLATE utf8mb4_unicode_ci NOT NULL,
  `subject` enum('matemática','português') COLLATE utf8mb4_unicode_ci NOT NULL,
  `theme` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL,
  `difficulty` enum('easy','medium','hard') COLLATE utf8mb4_unicode_ci NOT NULL DEFAULT 'medium',
  `year` int DEFAULT NULL,
  `statement` text COLLATE utf8mb4_unicode_ci NOT NULL,
  `alternatives` json NOT NULL,
  `correct_answer` char(1) COLLATE utf8mb4_unicode_ci NOT NULL,
  `explanation` text COLLATE utf8mb4_unicode_ci,
  `source` enum('enem_real_2009_2023','generated_system') COLLATE utf8mb4_unicode_ci DEFAULT 'enem_real_2009_2023',
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `questions_type_subject_theme_index` (`type`,`subject`,`theme`)
) ENGINE=InnoDB AUTO_INCREMENT=1152 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci

--- TABLE: simulation_answers ---
CREATE TABLE `simulation_answers` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `simulation_id` bigint unsigned NOT NULL,
  `question_id` bigint unsigned NOT NULL,
  `user_answer` char(1) COLLATE utf8mb4_unicode_ci DEFAULT NULL,
  `is_correct` tinyint(1) NOT NULL DEFAULT '0',
  `time_spent` int NOT NULL DEFAULT '0',
  `marked_for_review` tinyint(1) NOT NULL DEFAULT '0',
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `simulation_answers_simulation_id_question_id_unique` (`simulation_id`,`question_id`),
  KEY `simulation_answers_question_id_foreign` (`question_id`),
  CONSTRAINT `simulation_answers_question_id_foreign` FOREIGN KEY (`question_id`) REFERENCES `questions` (`id`) ON DELETE CASCADE,
  CONSTRAINT `simulation_answers_simulation_id_foreign` FOREIGN KEY (`simulation_id`) REFERENCES `simulations` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=871 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci

--- TABLE: simulations ---
CREATE TABLE `simulations` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `user_id` bigint unsigned NOT NULL,
  `type` enum('enem','concurso') COLLATE utf8mb4_unicode_ci NOT NULL,
  `configuration` json NOT NULL,
  `started_at` timestamp NULL DEFAULT NULL,
  `finished_at` timestamp NULL DEFAULT NULL,
  `time_elapsed` int NOT NULL DEFAULT '0',
  `status` enum('pending','in_progress','finished','corrected') COLLATE utf8mb4_unicode_ci NOT NULL DEFAULT 'pending',
  `score` decimal(5,2) DEFAULT NULL,
  `scores_by_subject` json DEFAULT NULL,
  `analysis_by_theme` json DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `simulations_user_id_status_index` (`user_id`,`status`),
  KEY `simulations_created_at_index` (`created_at`),
  CONSTRAINT `simulations_user_id_foreign` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=21 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
